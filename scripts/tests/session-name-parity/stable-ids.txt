# scripts/check-session-name-parity.sh の入力。1 行 1 件の安定 ID
# (`git rev-parse --absolute-git-dir` の出力そのまま) を並べる。
# 空行と `#` で始まる行は無視する (安定 ID は必ず `/` で始まるので曖昧にならない)。
#
# 件を減らすときは、その形を落としてよいか考えること。ここに並んでいるのは
# bash 実装と Swift 実装がずれうる点そのもの (Unicode スカラ / バイト / 書記素の違い、
# 空要素の扱い、`_git` の分岐、32 スカラの切り詰め位置) である。

# 通常の worktree

/Users/dev/repo/.git
/.git
# bare repository の <repo>.git は repo_git になり、_git の分岐に入らない
/Users/dev/repo.git
# 実在する _git という名前のディレクトリ (Project Root ではないが同じ分岐を通る)
/Users/dev/repo/_git
/_git

# 空要素になる形 (Swift の split は空要素を落とす)
/

# 記号。生成名に . と : が出てはならない (TmuxSessionName の注記)
