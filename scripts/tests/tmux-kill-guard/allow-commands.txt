# ここから下は敵対的レビュー (#314) M3。事故を記録する文書・Issue 本文・コード例を heredoc で
# 書く経路は日常的に使う。ここを塞ぐと、当たったエージェントは逃し道が無く停止する。
cat > docs/note.md <<'EOF'

# C1 の修正が、行継続を挟んだ正当な後始末まで巻き込んでいないこと
env -u TMUX \
  tmux -L awt-x \
  kill-session -t '=awt-x'
%%
# 予約語の読み飛ばしが、隔離できている後始末を通すこと (M1)
if tmux -L awt-x has-session -t '=awt-x'; then tmux -L awt-x kill-session -t '=awt-x'; fi
%%
# M-A の修正が、末尾リダイレクトを伴う正当な後始末を巻き込んでいないこと
env -u TMUX tmux -L awt-x kill-session -t "=awt-x" 2>/dev/null
%%

%%
tmux -L awt-x kill-session -t "=awt-x" >> /tmp/log 2>&1
%%
# M-B: シェルに文字列やファイルを食わせる形は検査しない (決定)。`bash /tmp/x.sh` が
# スコープ外である以上、ここだけ塞いでも境界にならないため、通ることを固定する。
bash <<EOF

%%

%%
cat <<'EOF' | bash

%%
echo 'tmux kill-server' | sh
%%
# m2: escape された展開は bash も展開しないので拒否しない
cat <<EOF

EOF
%%
# M-1 の修正が、正当な後始末の &> や >| を巻き込んでいないこと

%%
tmux -L awt-x kill-session -t "=awt-x" >|/tmp/o
%%
